    <script>
        // --- 1. Bioscience Mathematical Model ---
        const L = 75;    // Maximum oil content (~75% of dry mesocarp weight)
        const k = 1.2;   // Logistic growth steepness
        const x0 = 16;   // Midpoint of growth (16 Weeks After Anthesis)

        // Generate full dataset
        const rawData = [];
        for (let x = 10; x <= 22; x += 0.5) {
            // Function f(x): Logistic growth
            const fx = L / (1 + Math.exp(-k * (x - x0)));
            
            // Derivative f'(x): Rate of change using chain/quotient rule
            const dfx = (L * k * Math.exp(-k * (x - x0))) / Math.pow(1 + Math.exp(-k * (x - x0)), 2);
            
            rawData.push({ x: x, y: fx, derivative: dfx });
        }

        // --- 2. Chart.js Initialization ---
        const ctx = document.getElementById('growthChart').getContext('2d');
        
        const chart = new Chart(ctx, {
            type: 'line',
            data: {
                datasets: [
                    {
                        label: 'Total Oil Content (%) f(x)',
                        data: [], // Will be populated by animation
                        borderColor: '#f59e0b', // amber-500
                        backgroundColor: 'rgba(245, 158, 11, 0.1)',
                        borderWidth: 3,
                        pointBackgroundColor: '#fff',
                        pointBorderColor: '#d97706',
                        pointBorderWidth: 2,
                        pointRadius: 4,
                        pointHoverRadius: 8,
                        pointHoverBackgroundColor: '#f59e0b',
                        fill: true,
                        tension: 0.4 // Smooth curve
                    },
                    {
                        label: "Tangent Line f'(x)",
                        data: [], // Populated on click
                        borderColor: '#10b981', // emerald-500
                        borderWidth: 2,
                        borderDash: [5, 5],
                        pointRadius: 0,
                        tension: 0
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: {
                    mode: 'nearest',
                    axis: 'x',
                    intersect: true
                },
                scales: {
                    x: {
                        type: 'linear',
                        title: { display: true, text: 'Time (Weeks After Anthesis - WAA)', font: {weight: 'bold'} },
                        min: 9.5,
                        max: 22.5,
                        grid: { color: '#f1f5f9' }
                    },
                    y: {
                        title: { display: true, text: 'Mesocarp Oil (% Dry Weight)', font: {weight: 'bold'} },
                        min: 0,
                        max: 85,
                        grid: { color: '#f1f5f9' }
                    }
                },
                plugins: {
                    tooltip: {
                        callbacks: {
                            label: (context) => {
                                if (context.datasetIndex === 1) return null; // Hide tooltip for tangent
                                return `Oil Content: ${context.parsed.y.toFixed(2)}%`;
                            }
                        }
                    }
                },
                onClick: (evt, activeElements) => handleChartClick(activeElements)
            }
        });

        // --- 3. Progressive Drawing Animation ---
        let currentIndex = 0;
        const animationSpeed = 40; // ms per point

        const drawingInterval = setInterval(() => {
            if (currentIndex < rawData.length) {
                chart.data.datasets[0].data.push({
                    x: rawData[currentIndex].x,
                    y: rawData[currentIndex].y
                });
                chart.update('none'); // Update without full animation re-render
                currentIndex++;
            } else {
                clearInterval(drawingInterval);
                // Reveal interaction hint
                const hint = document.getElementById('interaction-hint');
                hint.classList.remove('opacity-0');
                hint.classList.add('animate-pulse-text');
            }
        }, animationSpeed);


        // --- 4. Interactive Derivative Analysis (The Core Feature) ---
        function handleChartClick(activeElements) {
            if (!activeElements || activeElements.length === 0) return;

            // Get the clicked data point
            const dataIndex = activeElements[0].index;
            const pointData = rawData[dataIndex];
            
            const x_c = pointData.x;
            const y_c = pointData.y;
            const slope = pointData.derivative; // This is f'(x)

            // Draw Tangent Line: y - y_c = m(x - x_c)  =>  y = m(x - x_c) + y_c
            // Create a line segment 3 units wide centered on the click
            const span = 1.5; 
            const tangentData = [
                { x: x_c - span, y: slope * ((x_c - span) - x_c) + y_c },
                { x: x_c + span, y: slope * ((x_c + span) - x_c) + y_c }
            ];

            chart.data.datasets[1].data = tangentData;
            chart.update();

            // Update DOM Analysis Panel
            document.getElementById('val-week').textContent = x_c.toFixed(1);
            document.getElementById('val-oil').textContent = y_c.toFixed(1);
            document.getElementById('val-rate').textContent = slope.toFixed(2);

            // Generate Bioscience / Agro-Tech explanation based on the derivative
            let explanation = "";
            
            if (x_c < 14) {
                explanation = "Lag Phase: The mesocarp is mostly water and carbohydrates right now. The rate of oil synthesis (derivative) is very low. Harvesting now yields almost zero commercial oil.";
            } else if (x_c >= 14 && x_c <= 17.5) {
                explanation = "Exponential Phase: Look at that steep tangent line! The derivative is highly positive. The plant is actively converting sugars into lipids. Never harvest during this massive yield-building window.";
            } else if (x_c > 17.5 && x_c < 20) {
                explanation = "Deceleration Phase: The rate of change is dropping. The fruit's 'oil tank' is nearly full. Mill processing engineers start preparing for this incoming batch.";
            } else {
                explanation = "Plateau / Optimum Harvest: The derivative f'(x) is practically zero, meaning oil accumulation has stopped. This is the optimal window to harvest FFB before fruit detachment drops loose fruits on the ground and Free Fatty Acids (FFA) degrade oil quality.";
            }

            document.getElementById('val-explanation').textContent = explanation;

            // Reveal and animate the panel
            const panel = document.getElementById('analysis-panel');
            panel.classList.remove('hidden');
            // Small delay to allow display:block to apply before animating transform
            setTimeout(() => {
                panel.classList.remove('translate-y-4', 'opacity-0');
            }, 10);
        }
    </script>
